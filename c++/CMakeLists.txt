find_package(Boost REQUIRED CONFIG)

target_include_directories(_core PRIVATE ${Boost_INCLUDE_DIRS})

target_include_directories(_test PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(_test PRIVATE gccjit)
install(TARGETS _core _test DESTINATION ${SKBUILD_PROJECT_NAME})
